\begin{lemma}[The potential of a norm ball]\label{lem:ballpotential}
Let $\gauge$ be a norm, let $\rho>0$, and let $y\in\R^d$. Then
\begin{equation}\label{eq:ballpotential}
U_{\{\gauge<\rho\}}(y)=2d\drift(\rho-\gauge(y))_+\, .
\end{equation}
\end{lemma}
\end{verbatim}
