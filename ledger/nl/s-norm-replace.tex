\paragraph{Statement.} $|\sum_{x\in A_n}|x|-\int_{D_n}|v|\dd v|\leq\frac{\sqrt d}2|A_n|$.
\paragraph{Proof.} $\int_{D_n}|v|=\sum_{x\in A_n}\int_{C_x}|v|$ (disjoint cells), $|C_x|=1$, and $||v|-|x||\leq|v-x|\leq\sqrt d/2$ on $C_x$.
