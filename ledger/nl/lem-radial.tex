\paragraph{Paper (verbatim).}
\begin{lemma}\label{lem:radial}
Let $p>0$. There exist $\rho_0(d,\gauge)>8d$ and $C(d,\drift,\gauge,p)<\infty$ such that, for every integer~$n\geq2$, with probability at least $1-Cn^{-p}$, the following holds simultaneously for all integers~$\rho_0\leq\rho\leq n$:
\begin{equation}\label{eq:radial}
\begin{split}
F(\rho+4d)\leq{}&C\Bigl(\max_{||x|-\rho|\leq3}\ell_n(x)\Bigr)
[F(\rho-4d)-F(\rho+4d)]\\
&+C\Bigl(\max_{|x|\geq\rho-4d}\ell_n(x)\cdot\rho^{1-d}F(\rho-4d)\log n\Bigr)^{\nf12}+C\rho^{1-d}\log n\, .
\end{split}
\end{equation}
\end{lemma}
Definitions the statement relies on (verbatim): the weighted exterior volume $F$ (eq:Fdef, lines 503--506) and the
constants convention (line 295).
\begin{equation}\label{eq:Fdef}
F(s)\coloneqq\frac1{d\omega_d}\int_{D\cap\{|v|>s\}}|v|^{1-d}\dd v
\leq\frac{|D|}{d\omega_ds^{d-1}}\, .
\end{equation}
\begin{itemize}
\item The letters $c>0$ and $C>0$ denote constants depending only on $d$, $\drift$ and the exponent~$p$ of the failure probability under discussion, and in Sections~\ref{sec:norm-setup}--\ref{sec:norms} also on the norm~$\gauge$, but not on the choice of the subgradients~$\xi(x)$; they may change from line to line. Every further dependence is stated, and a subscript on~$O$, as in~$O_y$, means that the implied constant also depends on~$y$. We write $a_n\asymp b_n$ if $c\leq a_n/b_n\leq C$, and $O_{\P}(a_n)$ for random variables~$Y_n$ such that $Y_n/a_n$ is tight.
\end{itemize}
\paragraph{Transcription.}
\begin{itemize}
\item \emph{Quantifier order.} $\rho_0(d,\gauge)>8d$ is bound right after $\gauge$ and before $\drift$, $p$, $\xi$, the walk:
 \texttt{exists rho0 : R, 8*d < rho0 /\ forall eps, \dots}. $C(d,\drift,\gauge,p)$ is bound after $p$ and before $\xi$ and
 the probability space (not depending on the subgradients, line 295). $\rho_0$ is real; ``integers $\rho_0\le\rho\le n$'' is
 \texttt{forall rho : N, rho0 <= rho -> rho <= n -> \dots}.
\item \emph{Setting.} As in the coarse-bounds twin: \texttt{IsNorm Psi}, $\drift>0$ with eq:ellipticity, subgradients $\xi$ with
 $\xi(0)=0$, \texttt{IsDriftCERW mu eps xi X}, every $n\ge2$.
\item \emph{F.} ``$F$ is the function eq:Fdef with $D=D_n$'' is \texttt{CERW.tail d (CERW.cellSet (X . w) n)}, with
 $\mathrm{tail}(D,s)=(d\,\omega_d)^{-1}\int_{D\cap\{|v|>s\}}\|v\|^{1-d}\,dv$ ($\sigma_d=d\omega_d$, $\omega_d=$ \texttt{unitBallVolume d}).
 Arguments are $(\rho:\mathbb R)\pm4d$, which are positive because $\rho>8d$.
\item \emph{Shell maximum.} $\max_{||x|-\rho|\le3}\ell_n(x)=$ \texttt{CERW.shellMax (X . w) n rho} (a \texttt{Finset.sup} over the
 lattice points with $||x|-\rho|\le3$), cast to \texttt{R}.
\item \emph{Tail maximum.} $\max_{|x|\ge\rho-4d}\ell_n(x)$ is the \texttt{Finset.sup} of \texttt{localTime} over the sites of
 \texttt{departureRange} with $\rho-4d\le$ \texttt{euclidNorm x}; since $\ell_n=0$ off $A_n$ this is the maximum over all of $\Z^d$
 (and $0$ if empty, which is the value of the maximum of zeros).
\item \emph{Three terms.} The displayed inequality is $F(\rho+4d)\le C\,M_{\rm sh}\,[F(\rho-4d)-F(\rho+4d)]
 +C\sqrt{M_{\ge}\,\rho^{1-d}F(\rho-4d)\log n}+C\rho^{1-d}\log n$, with $\rho^{1-d}$ a real power and
 $\log n=$ \texttt{Real.log n}. All three terms carry the same $C$, which also bounds the failure probability
 (maximum of the line-by-line constants; all three coefficients multiply nonnegative quantities).
\item \emph{Event.} $\mu\{\omega:\neg\forall\rho\dots\}\le\texttt{ENNReal.ofReal}(C\,n^{-p})$ (outer measure).
\end{itemize}
