\paragraph{Paper (verbatim).}
\begin{lemma}\label{lem:geometry}
Let $D\subset\R^d$ be a bounded measurable set. There exists $C(d,\gauge)<\infty$ such that the following hold.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \underline{\emph{Bound}}:
\begin{equation}\label{eq:potential-bound}
\sup_{y\in\R^d}|U_D(y)|\leq C\drift|D|^{\nf{1}{d}}\, .
\end{equation}
\item \underline{\emph{H\"older continuity}}: for all $y,z\in\R^d$,
\begin{equation}\label{eq:holder}
|U_D(y)-U_D(z)|\leq C\drift|D|^{\nf{1}{2d}}|y-z|^{\nf{1}{2}}\, .
\end{equation}
\item \underline{\emph{Spherical averages}}: for every $s>0$,
\begin{equation}\label{eq:newton}
\frac1{d\omega_ds^{d-1}}\int_{\{|v|=s\}}U_D(v)\dd S(v)
=\frac{2\drift}{\omega_d}\int_{D\cap\{|v|>s\}}\frac{\gauge(v)}{|v|^d}\dd v\, .
\end{equation}
The right side of~\eqref{eq:newton} lies between $2d\drift c_\gauge F(s)$ and $2d\drift\Lambda_\gauge F(s)$, and it equals $2d\drift F(s)$ for the Euclidean norm.
\end{enumerate}
\end{lemma}
Definitions the statement relies on (verbatim):
\textit{eq:Fdef (lines 503--507)}
\begin{equation}\label{eq:Fdef}
F(s)\coloneqq\frac1{d\omega_d}\int_{D\cap\{|v|>s\}}|v|^{1-d}\dd v
\leq\frac{|D|}{d\omega_ds^{d-1}}\, .
\end{equation}
The function~$F$ is nonincreasing and continuous. We write $\dd S$ for surface measure on a sphere.
\textit{$\Lambda_\gauge$, $c_\gauge$ (lines 327--327)}
In Sections~\ref{sec:norm-setup}--\ref{sec:norms}, $\gauge$ is a norm, $\drift$ satisfies~\eqref{eq:ellipticity}, $r_n$ is given by~\eqref{eq:radius-norm}, and $U_D$ is the potential~\eqref{eq:potential-norm} defined below; at a point~$v\in\R^d$, $\nabla\gauge(v)$ is the gradient, which exists for almost every~$v$. Let $\Lambda_\gauge\coloneqq\max_{|u|=1}\gauge(u)$ and $c_\gauge\coloneqq\min_{|u|=1}\gauge(u)$; every $\xi\in\partial\gauge(x)$ satisfies $\xi\cdot x=\gauge(x)$ and $|\xi|\leq\Lambda_\gauge$. In Sections~\ref{sec:contact}--\ref{sec:lower}, $\gauge$ is the Euclidean norm, so that $\xi(x)=u_x$, $r_n$ is given by~\eqref{eq:radius}, and $U_D$ is the potential~\eqref{eq:potential-intro}.
\textit{eq:potential-norm (lines 348--352)}
For a norm~$\gauge$, the potential~\eqref{eq:potential-intro} is defined with the gradient of~$\gauge$ in place of~$v/|v|$: for a bounded measurable set~$D\subset\R^d$ and $y\in\R^d$, let
\begin{equation}\label{eq:potential-norm}
U_D(y)\coloneqq\frac{2\drift}{\omega_d}\int_D\nabla\gauge(v)\cdot\frac{v-y}{|v-y|^d}\dd v
\end{equation}
be the potential generated by the vector field~$-\drift\nabla\gauge$ on~$D$; for the Euclidean norm it is~\eqref{eq:potential-intro}.
\paragraph{Transcription.}
Paper macros: \verb!\drift! is $\kappa$ (Lean \verb!ε!), \verb!\gauge! is $\Psi$, \verb!\nf{a}{b}! is $a/b$.

\emph{Object mapping.}
\begin{itemize}
\item $\omega_d$ is \verb!ωd := (volume (Metric.ball 0 1)).toReal! (positive, finite). $D$ is \verb!D : Set (EuclideanSpace ℝ (Fin d))! with \verb!MeasurableSet D! and \verb!Bornology.IsBounded D!; $|D|$ is \verb!R := (volume D).toReal! (finite because $D$ is bounded).
\item $U_D$ is \verb!CERW.normPotential d ε Ψ D! (see lem-ballpotential for the gradient convention and the integrability of the integrand); $F(s)$ is \verb!CERW.tail d D s! $=(d\,\omega_d)^{-1}\int_{D\cap\{\|v\|>s\}}\|v\|^{1-d}\,dv$ (eq:Fdef); $\Lambda_\gauge,c_\gauge$ are \verb!CERW.normMax Ψ! $=\sup\gauge|_{S^{d-1}}$ and \verb!CERW.normMin Ψ! $=\inf\gauge|_{S^{d-1}}$, which are attained (the sphere is nonempty and compact, $\gauge$ is continuous), so \verb!sSup!/\verb!sInf! are the true maximum and minimum.
\item The surface measure $\dd S$ on $\{|v|=s\}$ and the normalisation $\tfrac1{d\omega_ds^{d-1}}\int_{\{|v|=s\}}(\cdot)\,\dd S$ are written as the average over the unit sphere: \verb!(d * ωd)⁻¹ * ∫ θ, normPotential d ε Ψ D (s • θ) ∂(volume).toSphere!, where \verb!Measure.toSphere! has total mass $d\,\omega_d$ (\verb!MeasureTheory.Measure.toSphere_apply_univ!) and realises the polar-coordinates formula, so this is the mean value of $U_D$ over the sphere of radius $s$. This is the normalisation of the already-frozen Euclidean \verb!CERW.Frozen.potential_geometry!. The integrand $\theta\mapsto U_D(s\theta)$ is continuous (it is Hölder by (ii)), so the integral is genuine.
\item The right side of eq:newton is \verb!B := 2 * ε / ωd * ∫ v in D ∩ {v | s < ‖v‖}, Ψ v / ‖v‖ ^ d! (bounded integrand on a bounded set, $\|v\|>s>0$, so integrable and no division by zero); $A=B$ is eq:newton.
\end{itemize}

\emph{Reading choices.}
\begin{enumerate}
\item \textbf{Quantifier order.} \verb!∀ Ψ, IsNorm Ψ → ∃ C, 0 < C ∧ ∀ ε, 0 < ε → ∀ D, MeasurableSet D → IsBounded D → ..!. The paper writes ``Let $D$ be a bounded measurable set. There exists $C(d,\gauge)$''; the notation $C(d,\gauge)$ says that $C$ does not depend on $D$, and the lemma is used uniformly over $D_n$, so $C$ is chosen after $\Psi$ and before $\varepsilon$ and $D$. The parameter $\kappa=$ \verb!ε! only enters through the explicit factor in each bound and through the linearity of $U_D$ in $\kappa$: putting $\exists C$ before $\forall\varepsilon$ loses nothing (machine-checked for (i): \verb!sup_bound_all_eps!).
\item \textbf{$\varepsilon$.} The paper's $\kappa$ satisfies the standing ellipticity condition; the lemma uses only $\kappa>0$ (for the orientation of the inequalities in (iii)), so the draft binds every $\varepsilon>0$. This is a true strengthening by linearity in $\kappa$.
\item \textbf{The three clauses.} (i) is \verb!∀ y, |normPotential d ε Ψ D y| ≤ C * ε * R ^ ((1 : ℝ) / d)! (the supremum is written $\forall y$); (ii) is \verb!∀ y z, |..y - ..z| ≤ C * ε * R ^ ((1 : ℝ) / (2 * d)) * ‖y - z‖ ^ ((1 : ℝ) / 2)!; (iii) is, for every $s>0$, the conjunction of the four statements below. When $R=0$ both sides of (i) are $0$ ($D$ null, $U_D=0$).
\item \textbf{(iii) in four conjuncts.} \verb!A = B! is eq:newton; \verb!2 * d * ε * normMin Ψ * tail d D s ≤ B! and \verb!B ≤ 2 * d * ε * normMax Ψ * tail d D s! say that the right side lies between $2d\kappa c_\gauge F(s)$ and $2d\kappa\Lambda_\gauge F(s)$; and \verb!(∀ v, Ψ v = ‖v‖) → B = 2 * d * ε * tail d D s! is ``it equals $2d\kappa F(s)$ for the Euclidean norm'', read as the case in which $\gauge$ is the Euclidean norm.
\item Euclidean consistency (machine-checked): for $\gauge=|\cdot|$, \verb!normPotential = potential!, so the clauses specialise to the proved Euclidean lemma.
\item Numerical check of the statement as written ($d=2$, $\gauge=\|\cdot\|_\infty$, $D=[-1,1]^2$, $\varepsilon=1$, $s=\tfrac12$): $A=2.19937$, $B=2.19934$, and $2d\kappa c_\gauge F=1.7598\le B=2.1993\le2d\kappa\Lambda_\gauge F=2.4888$ with $F=0.62219$, $c_\gauge=2^{-1/2}$, $\Lambda_\gauge=1$.
\end{enumerate}
