% s-var-high: eq:masshigh and the envelope with rate W (deterministic, d >= 3).
\paragraph{Statement.} For $d\geq3$, $\varepsilon>0$, $K\geq2$, $C_0,C_1\geq0$ there is $C>0$ such that: if $n\geq1$, $|X_j|\leq n$ for $j<n$, $L=\log(n+2)\leq N$, $b>0$,
$B(0,b)\subseteq D_n\subseteq B(0,(K-1)N)$, $|y_0|=b$, $\ell_n(z)=0$, $b\leq|z|\leq n$, $|U(y_0)-U(z)|\leq\Delta\leq C_0L$, and for all sites $|y|\leq n$:
$|\ell_n(y)-U(y)+M(y)|\leq C_1L$ and $|M(y)|\leq C_1(\sqrt{B_yL}+L)$ with $B_y=\sum_{x\in A_n}\ell_n(x)(1+|x-y|)^{2-2d}$, then
$m\leq CN^dQ$ and $\ell_n(y)\leq C(b-|y|)_++CNQ^{1/d}$ for all sites, where $Q=(L/N)^{d/(2d-1)}$.
\paragraph{Proof.} (fl:912--969.) (1) \texttt{localTime\_le\_envelope} with $\rho=\ell_n-U+M$ at each $y\in A_n$: $\ell_n(y)\leq c_0s_b(y)+A_1+C_1\sqrt{B_yL}$,
$c_0=2d\varepsilon$, $A_1=c_0\omega_d^{-1/d}m^{1/d}+2C_1L$. (2) \texttt{exists\_envelope\_high} ($A_2=C_1$): $\ell_n\leq C(c_0s_b+A_1+C_1^2L+c_0L)$ everywhere and
$B_z\leq C(A_1+C_1^2L+c_0L)\leq C(m^{1/d}+L)$. (3) \texttt{volume\_sdiff\_le\_contact} at $z$ with $S=(K-1)N$: $m\leq CN^{d-1}(C_1L+C_1\sqrt{B_zL}+C_1L+C_0L)
\leq CN^{d-1}(m^{1/(2d)}\sqrt L+L)$. \texttt{le\_of\_le\_mul\_rpow\_mul\_sqrt\_add}: $m\leq C(N^{2d(d-1)/(2d-1)}L^{d/(2d-1)}+N^{d-1}L)$; the first term is $N^dQ$ and
$N^{d-1}L\leq N^dQ$ iff $L\leq N$. (4) $m^{1/d}\leq CNQ^{1/d}$ and $L\leq NQ^{1/d}=N(L/N)^{1/(2d-1)}$ since $L\leq N$.
