% s-contact-lower: the lower bound of eq:contactbound.
\paragraph{Statement.} If $B(0,b)\subseteq D\subseteq\bar B(0,S)$, $D$ measurable, $|y_0|=b$, $\eps\geq0$, then
$U_D(y_0)\geq\frac{2\eps}{\omega_d}2^{1-d}S^{1-d}|D\setminus B(0,b)|$.
\paragraph{Proof.} Split $D=B(0,b)\sqcup E$ with $E=D\setminus B(0,b)$ (both integrands are integrable, s-potential-bound). $U_{B(0,b)}(y_0)=2d\eps(b-|y_0|)_+=0$
(s-newton (iii); for $b=0$ the ball is empty). For $v\in E$, $|v|\geq b=|y_0|$ and $v\neq y_0$ a.e., so the integrand is
$\geq2^{1-d}|v|^{1-d}\geq2^{1-d}S^{1-d}$ (s-contact-kernel, $|v|\leq S$).
