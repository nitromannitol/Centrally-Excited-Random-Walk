% s-shellW: eq:shellW, and the error-scale bounds of eq:global.
\paragraph{Statement.} (i) If $\ell_n\leq c_0(b-|\cdot|)_++A$ everywhere with $A\geq0$ and $r\geq b+3$, then $M_{\rm sh}(r)\leq A$. (ii) If $0\leq M\leq C_0N$ with $N,L\geq1$, then
$\sqrt ML+L\leq(\sqrt{C_0}+1)\sqrt NL$ and $\sqrt{ML}+L\leq(\sqrt{C_0}+1)(\sqrt{NL}+L)$.
\paragraph{Proof.} (i) Shell sites have $|y|\geq r-3\geq b$, so $(b-|y|)_+=0$. (ii) $\sqrt M\leq\sqrt{C_0}\sqrt N$ and $1\leq\sqrt N$.
