\paragraph{Paper excerpt.}
\begin{verbatim}
\begin{lemma}\label{lem:layer}
Let $b\geq0$ and $a>0$, and let $D$ be a measurable subset of $\{v\in\R^d:b\leq \gauge(v)\leq b+a\}$. Then
\begin{equation*}
\sup_{y\in\R^d}|U_D(y)|\leq2d\drift a\, .
\end{equation*}
\end{lemma}
\end{verbatim}

\paragraph{Transcription.}
\begin{itemize} \item \emph{Setting.} $d\ge2$ is \texttt{hd}; $\gauge$ is \texttt{Psi} with \texttt{IsNorm Psi}; $\drift$ is \texttt{eps} with $0<\drift$. The declaration does not assume eq:ellipticity: the proof (polar coordinates about $y$, convexity of $t\mapsto\gauge(y+tu)$, change of variables $s=h_u(t)$) never uses it, so this is the paper's lemma for a wider class of $\drift$. $\drift>0$ is needed, because the right side $2d\drift a$ would be negative otherwise. There are no constants and no probability space. \item \emph{Shell.} $b\ge0$ and $a>0$ are reals (\texttt{hb}, \texttt{ha}). $D$ is a set of \texttt{EuclideanSpace R (Fin d)} with \texttt{MeasurableSet D} (Borel, the repository's reading of ``measurable'', as in \texttt{potential\_geometry}) and \texttt{D $\subseteq$ \{v | b <= Psi v /\ Psi v <= b + a\}}.
