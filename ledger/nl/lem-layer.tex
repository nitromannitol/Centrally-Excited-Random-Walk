\paragraph{Paper (verbatim).}
\begin{lemma}\label{lem:layer}
Let $b\geq0$ and $a>0$, and let $D$ be a measurable subset of $\{v\in\R^d:b\leq \gauge(v)\leq b+a\}$. Then
\begin{equation*}
\sup_{y\in\R^d}|U_D(y)|\leq2d\drift a\, .
\end{equation*}
\end{lemma}
Definitions the statement relies on (verbatim): the potential of a norm (eq:potential-norm, lines 357--361) and the
notation $\Lambda_\gauge,c_\gauge$ (line 336).
For a norm~$\gauge$, the potential~\eqref{eq:potential-intro} is defined with the gradient of~$\gauge$ in place of~$v/|v|$: for a bounded measurable set~$D\subset\R^d$ and $y\in\R^d$, let
\begin{equation}\label{eq:potential-norm}
U_D(y)\coloneqq\frac{2\drift}{\omega_d}\int_D\nabla\gauge(v)\cdot\frac{v-y}{|v-y|^d}\dd v
\end{equation}
be the potential generated by the vector field~$-\drift\nabla\gauge$ on~$D$; for the Euclidean norm it is~\eqref{eq:potential-intro}.
\begin{quote}
In Sections~\ref{sec:norm-setup}--\ref{sec:norms}, $\gauge$ is a norm, $\drift>0$ satisfies~\eqref{eq:ellipticity}, $r_n$ is given by~\eqref{eq:radius-norm}, and $U_D$ is the potential~\eqref{eq:potential-norm} defined below; at a point~$v\in\R^d$, $\nabla\gauge(v)$ is the gradient, which exists for almost every~$v$. Let $\Lambda_\gauge\coloneqq\max_{|u|=1}\gauge(u)$ and $c_\gauge\coloneqq\min_{|u|=1}\gauge(u)$; every $\xi\in\partial\gauge(x)$ satisfies $\xi\cdot x=\gauge(x)$ and $|\xi|\leq\Lambda_\gauge$. In Sections~\ref{sec:contact}--\ref{sec:lower}, $\gauge$ is the Euclidean norm, so that $\xi(x)=u_x$, $r_n$ is given by~\eqref{eq:radius}, and $U_D$ is the potential~\eqref{eq:potential-intro}.
\end{quote}
\paragraph{Transcription.}
\begin{itemize}
\item \emph{Setting.} $d\ge2$ is \texttt{hd}; $\gauge$ is \texttt{Psi} with \texttt{IsNorm Psi}; $\drift$ is \texttt{eps} with
 $0<\drift$. The declaration does not assume eq:ellipticity: the proof (polar coordinates about $y$, convexity of
 $t\mapsto\gauge(y+tu)$, change of variables $s=h_u(t)$) never uses it, so this is the paper's lemma for a wider class of $\drift$.
 $\drift>0$ is needed, because the right side $2d\drift a$ would be negative otherwise. There are no constants and no
 probability space.
\item \emph{Shell.} $b\ge0$ and $a>0$ are reals (\texttt{hb}, \texttt{ha}). $D$ is a set of \texttt{EuclideanSpace R (Fin d)} with
 \texttt{MeasurableSet D} (Borel, the repository's reading of ``measurable'', as in \texttt{potential\_geometry}) and
 \texttt{D $\subseteq$ \{v | b <= Psi v /\ Psi v <= b + a\}}. The shell is bounded (because $\gauge$ is a norm), so no boundedness
 hypothesis is needed.
\item \emph{Potential.} $U_D(y)=$ \texttt{CERW.normPotential d eps Psi D y}
 $=\frac{2\drift}{\omega_d}\int_D\langle\nabla\gauge(v),v-y\rangle/\|v-y\|^d\,dv$ with $\omega_d=$ \texttt{unitBallVolume d} and Mathlib's
 \texttt{gradient}, which is the gradient where $\gauge$ is differentiable and $0$ elsewhere (a null set by Rademacher, so the
 integral is the paper's). At $v=y$ the integrand reads $0$ (division by zero), a null set.
\item \emph{Conclusion.} $\sup_{y}|U_D(y)|\le2d\drift a$ is \texttt{forall y, |normPotential d eps Psi D y| <= 2 * d * eps * a}
 (a family of reals is bounded by $M$ iff each member is).
\end{itemize}
